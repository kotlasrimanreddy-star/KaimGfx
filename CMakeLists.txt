cmake_minimum_required(VERSION 3.20)
project(KaimGfxUI LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "KaimGfxUI is a Windows desktop application.")
endif()
add_executable(KaimGfx WIN32 main.cpp)
target_compile_features(KaimGfx PRIVATE cxx_std_17)
target_link_libraries(KaimGfx PRIVATE gdiplus user32 gdi32)
if(MSVC)
  target_compile_options(KaimGfx PRIVATE /W4 /utf-8)
endif()
